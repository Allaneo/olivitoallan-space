{{- /* Override of Blowfish's render-image.html. The theme version always emits 330w/660w/1280w in the srcset, even when the source image is narrower than 1280px. Hugo happily upscales, so a 500px-wide screenshot got interpolated up to 1280px and looked blurry on retina screens. Here the candidate widths are clamped to the source width, and the image's own native width is always offered as the largest candidate. A `sizes` hint is added so the browser stops assuming the image spans the full viewport. */ -}} {{ define "RenderImageSimple" -}} {{- end }} {{ define "RenderImageResponsive" -}} {{- $r := .resource }} {{- $native := $r.Width }} {{- $widths := slice }} {{- range (slice 330 660 1280) }} {{- if lt . $native }} {{- $widths = $widths | append . }} {{- end }} {{- end }} {{- $widths = $widths | append $native }} {{- end }} {{ define "RenderImageCaption" -}} {{- with .caption }} {{- end }} {{- end }} {{- $disableImageOptimizationMD := .Page.Site.Params.disableImageOptimizationMD | default false }} {{- $urlStr := .Destination | safeURL -}} {{- $url := urls.Parse $urlStr -}} {{- $altText := .Text }} {{- $caption := .Title }} {{- $isRemote := findRE "^(https?|data)" $url.Scheme }} {{- $resource := "" }} {{- if not $isRemote }} {{- $resource = or ($.Page.Resources.GetMatch $urlStr) (resources.Get $urlStr) }} {{- end }}
